cmake_minimum_required(VERSION 3.22)
project(dual_demosaic VERSION 0.5.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(ANDROID)
  set(VULKAN_LIB vulkan)
else()
  find_package(Vulkan REQUIRED)
  set(VULKAN_LIB Vulkan::Vulkan)
endif()
option(DUAL_USE_EXTERNAL_RCD "Link Dual against an already-defined canonical rcd::rcd target" ON)

if(DUAL_USE_EXTERNAL_RCD)
  if(NOT TARGET rcd::rcd)
    message(FATAL_ERROR "DUAL_USE_EXTERNAL_RCD=ON requires an existing rcd::rcd target")
  endif()
else()
  set(RCD_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  add_subdirectory(../rcd ${CMAKE_CURRENT_BINARY_DIR}/rcd_fallback EXCLUDE_FROM_ALL)
endif()

option(DUAL_USE_EXTERNAL_VNG4 "Link Dual against an already-defined canonical vng4::vng4 target" ON)

set(VNG4_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(VNG4_BUILD_BENCHMARK OFF CACHE BOOL "" FORCE)
if(DUAL_USE_EXTERNAL_VNG4)
  if(NOT TARGET vng4::vng4)
    message(FATAL_ERROR "DUAL_USE_EXTERNAL_VNG4=ON requires an existing vng4::vng4 target")
  endif()
elseif(NOT TARGET vng4::vng4)
  add_subdirectory(../vng4 ${CMAKE_CURRENT_BINARY_DIR}/vng4_fallback EXCLUDE_FROM_ALL)
endif()

if(NOT TARGET amaze::amaze)
  add_subdirectory(../amaze ${CMAKE_CURRENT_BINARY_DIR}/amaze_fallback EXCLUDE_FROM_ALL)
endif()

add_library(dual_demosaic STATIC src/Pipeline.cpp)
add_library(dual::dual_demosaic ALIAS dual_demosaic)
target_include_directories(dual_demosaic PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(dual_demosaic PUBLIC rcd::rcd vng4::vng4 amaze::amaze ${VULKAN_LIB})
target_compile_options(dual_demosaic PRIVATE $<$<CXX_COMPILER_ID:AppleClang,Clang,GNU>:-O3;-Wall;-Wextra;-Wpedantic;-Werror>)

option(DUAL_BUILD_VALIDATION "Build dual Vulkan validation harness" OFF)
if(DUAL_BUILD_VALIDATION)
  add_executable(dual_validate validation/dual_validate.cpp)
  target_link_libraries(dual_validate PRIVATE dual_demosaic ${VULKAN_LIB})
  add_executable(dual_benchmark validation/dual_benchmark.cpp)
  target_link_libraries(dual_benchmark PRIVATE dual_demosaic ${VULKAN_LIB})
endif()
