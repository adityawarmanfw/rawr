cmake_minimum_required(VERSION 3.22)
project(rexture_device_check LANGUAGES C CXX)
# NDK executable (and desktop build) that runs the Rexture merge on a rawburst folder: stage dumps for the NumPy comparison,
# timing against HDR+ Quality. Android: cmake -DCMAKE_TOOLCHAIN_FILE=$NDK/build/cmake/android.toolchain.cmake
#   -DANDROID_ABI=arm64-v8a -DANDROID_PLATFORM=android-29 -DGLSLC=$NDK/shader-tools/darwin-x86_64/glslc
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(RAWR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(NOT ANDROID)
  find_package(Vulkan REQUIRED)
endif()
if(NOT GLSLC)
  find_program(GLSLC glslc REQUIRED)
endif()
set(RAW_SHARPNESS_BUILD_VULKAN OFF CACHE BOOL "" FORCE)
set(RAW_SHARPNESS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${RAWR_ROOT}/native/multiframe" "${CMAKE_CURRENT_BINARY_DIR}/multiframe")
add_executable(rexture_device_check main.cpp)
target_include_directories(rexture_device_check PRIVATE "${RAWR_ROOT}/native/vk_common/testing" "${RAWR_ROOT}/tools/common")
if(ANDROID)
  target_link_libraries(rexture_device_check PRIVATE raw_gpu_pipeline vulkan log)
else()
  target_link_libraries(rexture_device_check PRIVATE raw_gpu_pipeline Vulkan::Vulkan)
endif()
