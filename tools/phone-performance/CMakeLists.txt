cmake_minimum_required(VERSION 3.24)
project(rawr_phone_performance LANGUAGES CXX)
include(CTest)
find_package(Threads REQUIRED)
get_filename_component(RAWR_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(APP "${RAWR_ROOT}/app/src/main/cpp")
if(NOT ANDROID)
  find_path(VULKAN_HEADERS vulkan/vulkan.h REQUIRED)
endif()
# Supply the Release build of RAWR's pinned libjpeg-turbo, not a system version.
set(TURBOJPEG_LIBRARY "" CACHE FILEPATH "Bundled libjpeg-turbo static library")
if(NOT EXISTS "${TURBOJPEG_LIBRARY}")
  message(FATAL_ERROR "Build native/libjpeg-turbo and pass -DTURBOJPEG_LIBRARY=<libturbojpeg.a>")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(RAWR_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(RAWR_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Werror -UNDEBUG)
add_compile_options($<$<CONFIG:Release>:-O2>)
add_library(rawr_strip_jpeg STATIC "${APP}/encoding/jpeg/StripJpegEncoder.cpp")
target_include_directories(rawr_strip_jpeg PUBLIC "${APP}" "${RAWR_ROOT}/native/libjpeg-turbo/src")
target_link_libraries(rawr_strip_jpeg PUBLIC "${TURBOJPEG_LIBRARY}" Threads::Threads)
add_executable(jpeg_benchmark jpeg_benchmark.cpp)
target_compile_definitions(jpeg_benchmark PRIVATE RAWR_SEGMENTED)
target_link_libraries(jpeg_benchmark PRIVATE rawr_strip_jpeg)
set(RAWR_JPEG_BASELINE_DIR "" CACHE PATH "Directory with baseline StripJpegEncoder.cpp/.h from git show")
if(RAWR_JPEG_BASELINE_DIR)
  add_executable(jpeg_benchmark_baseline jpeg_benchmark.cpp "${RAWR_JPEG_BASELINE_DIR}/StripJpegEncoder.cpp")
  target_include_directories(jpeg_benchmark_baseline PRIVATE "${APP}" "${RAWR_ROOT}/native/libjpeg-turbo/src")
  target_link_libraries(jpeg_benchmark_baseline PRIVATE "${TURBOJPEG_LIBRARY}" Threads::Threads)
endif()
add_executable(jpeg_strip_check "${RAWR_ROOT}/tools/jpeg_strip_check/jpeg_strip_check.cpp")
target_link_libraries(jpeg_strip_check PRIVATE rawr_strip_jpeg)
add_test(NAME jpeg_strip_check COMMAND jpeg_strip_check)
add_executable(jpeg_writer_check jpeg_writer_check.cpp
  "${APP}/encoding/jpeg/JpegCaptureWriter.cpp"
  "${APP}/diagnostics/logging/RuntimeTraceRecorder.cpp"
  "${RAWR_ROOT}/native/gainmap/src/UltrahdrMux.cpp")
target_include_directories(jpeg_writer_check PRIVATE "${RAWR_ROOT}/native/gainmap/include")
if(ANDROID)
  target_link_libraries(jpeg_writer_check PRIVATE log)
else()
  target_include_directories(jpeg_writer_check PRIVATE stubs "${VULKAN_HEADERS}")
endif()
target_link_libraries(jpeg_writer_check PRIVATE rawr_strip_jpeg)
add_test(NAME jpeg_writer_check COMMAND jpeg_writer_check)
set_tests_properties(jpeg_writer_check PROPERTIES ENVIRONMENT "TMPDIR=${CMAKE_CURRENT_BINARY_DIR}")
